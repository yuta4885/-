' 貼り付け先: 「ボード」シート（Sheet1）のコードウィンドウ
Option Explicit

Private Sub Worksheet_SelectionChange(ByVal Target As Range)
    Dim c As Range, st As Range, nm As String, msg As String

    If Target.Cells.Count <> 1 Then Exit Sub
    Set c = Target.Cells(1)

    ' I1 = 運用中／編集中 の切り替え
    If c.Address = "$I$1" Then
        If c.Value = "編集中" Then c.Value = "運用中" Else c.Value = "編集中"
        Call ResetSelection
        Exit Sub
    End If

    ' 編集中は何もしない（名前の変更やセルの移動ができる）
    If Me.Range("I1").Value = "編集中" Then Exit Sub

    ' 名前セル、または状態セル（名前の1つ下）をタッチしたとき
    If c.Value = "出勤" Or c.Value = "退勤" Then
        Set st = c
    ElseIf c.Offset(1, 0).Value = "出勤" Or c.Offset(1, 0).Value = "退勤" Then
        Set st = c.Offset(1, 0)
    Else
        Exit Sub
    End If

    nm = st.Offset(-1, 0).Value
    If st.Value = "退勤" Then msg = "出勤しますか？" Else msg = "退勤しますか？"

    ' 確認ダイアログ（はい＝切り替え／いいえ＝そのまま）
    If MsgBox(nm & " さん" & vbCrLf & vbCrLf & msg, vbYesNo + vbQuestion + vbDefaultButton2, "出勤退勤の確認") = vbYes Then
        Application.EnableEvents = False
        If st.Value = "退勤" Then st.Value = "出勤" Else st.Value = "退勤"
        Application.EnableEvents = True
        ' あいさつ（2秒で自動的に閉じる）
        If st.Value = "出勤" Then Call Greet("おはようございます") Else Call Greet("お疲れ様でした")
    End If

    ' 同じ場所をもう一度タッチしても反応するよう、選択を余白セルに逃がす
    Call ResetSelection
End Sub

Private Sub ResetSelection()
    Application.EnableEvents = False
    Me.Range("A1").Select
    Application.EnableEvents = True
End Sub

Private Sub Greet(ByVal txt As String)
    On Error Resume Next
    CreateObject("WScript.Shell").Popup txt, 2, "出勤退勤ボード", vbInformation
    On Error GoTo 0
End Sub
