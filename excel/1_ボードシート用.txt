' 貼り付け先: 「ボード」シート（Sheet1）のコードウィンドウ
Option Explicit

Private Sub Worksheet_SelectionChange(ByVal Target As Range)
    Dim c As Range, st As Range, nm As String, msg As String

    If Target.Cells.Count <> 1 Then Exit Sub
    Set c = Target.Cells(1)

    ' I1 = 運用中／編集中 の切り替え
    If c.Address = "$I$1" Then
        If c.Value = "編集中" Then c.Value = "運用中" Else c.Value = "編集中"
        Call ResetSelection
        Exit Sub
    End If

    ' 編集中は何もしない（名前の変更やセルの移動ができる）
    If Me.Range("I1").Value = "編集中" Then Exit Sub

    ' 名前セル、または状態セル（名前の1つ下）をタッチしたとき
    If c.Value = "出勤" Or c.Value = "退勤" Then
        Set st = c
    ElseIf c.Offset(1, 0).Value = "出勤" Or c.Offset(1, 0).Value = "退勤" Then
        Set st = c.Offset(1, 0)
    Else
        Exit Sub
    End If

    nm = st.Offset(-1, 0).Value
    If st.Value = "退勤" Then msg = "出勤しますか？" Else msg = "退勤しますか？"

    ' 確認ダイアログ（はい＝切り替え／いいえ＝そのまま）
    If MsgBox(nm & " さん" & vbCrLf & vbCrLf & msg, vbYesNo + vbQuestion + vbDefaultButton2, "出勤退勤の確認") = vbYes Then
        Application.EnableEvents = False
        If st.Value = "退勤" Then st.Value = "出勤" Else st.Value = "退勤"
        Application.EnableEvents = True
        ' あいさつ（1.5秒で自動的に消える）
        If st.Value = "出勤" Then Call Greet("おはようございます", True) Else Call Greet("お疲れ様でした", False)
    End If

    ' 同じ場所をもう一度タッチしても反応するよう、選択を余白セルに逃がす
    Call ResetSelection
End Sub

Private Sub ResetSelection()
    Application.EnableEvents = False
    Me.Range("A1").Select
    Application.EnableEvents = True
End Sub

Private Sub Greet(ByVal txt As String, ByVal isIn As Boolean)
    Const SECONDS As Single = 1.5
    Dim shp As Shape, t As Single

    On Error GoTo Done
    Application.EnableEvents = False
    Set shp = Me.Shapes.AddShape(msoShapeRoundedRectangle, 0, 0, 520, 110)
    With shp
        .Name = "GreetBox"
        .Left = (Me.Range("J1").Left - .Width) / 2
        .Top = Me.Range("A9").Top
        If isIn Then
            .Fill.ForeColor.RGB = RGB(185, 240, 198)
        Else
            .Fill.ForeColor.RGB = RGB(214, 47, 47)
        End If
        .Line.ForeColor.RGB = RGB(255, 255, 255)
        .Line.Weight = 4
        With .TextFrame2.TextRange
            .Text = txt
            .Font.Size = 40
            .Font.Bold = msoTrue
            If isIn Then .Font.Fill.ForeColor.RGB = RGB(18, 49, 28) Else .Font.Fill.ForeColor.RGB = RGB(255, 255, 255)
        End With
    End With

    ' 1.5秒表示（その間も画面は更新される）
    t = Timer
    Do While Timer - t < SECONDS And Timer >= t
        DoEvents
    Loop

Done:
    On Error Resume Next
    Me.Shapes("GreetBox").Delete
    Application.EnableEvents = True
End Sub
